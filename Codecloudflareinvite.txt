export default {
  async fetch(request, env) {
    const url = new URL(request.url);

    if (request.method === 'OPTIONS') {
      return new Response(null, { headers: corsHeaders(request) });
    }

    if (url.pathname === '/') {
      return new Response(HTML_CONTENT, {
        headers: {
          'content-type': 'text/html; charset=UTF-8',
          ...corsHeaders(request)
        }
      });
    }

    if (url.pathname === '/api/auth/discord/login') {
      const redirectUri = env.DISCORD_REDIRECT_URI || `${url.origin}/api/auth/discord/callback`;
      const discordUrl =
        `https://discord.com/api/oauth2/authorize?client_id=${encodeURIComponent(env.CLIENT_ID)}` +
        `&redirect_uri=${encodeURIComponent(redirectUri)}` +
        `&response_type=code&scope=identify`;
      return Response.redirect(discordUrl, 302);
    }

    if (url.pathname === '/api/auth/discord/callback') {
      const code = url.searchParams.get('code');
      if (!code) return json({ ok: false, error: 'Missing code' }, 400, request);

      try {
        const redirectUri = env.DISCORD_REDIRECT_URI || `${url.origin}/api/auth/discord/callback`;

        const tokenRes = await fetch('https://discord.com/api/oauth2/token', {
          method: 'POST',
          headers: { 'content-type': 'application/x-www-form-urlencoded' },
          body: new URLSearchParams({
            client_id: env.CLIENT_ID,
            client_secret: env.CLIENT_SECRET,
            grant_type: 'authorization_code',
            code,
            redirect_uri: redirectUri,
            scope: 'identify'
          })
        });

        const tokenText = await tokenRes.text();
        let tokenData = {};
        try { tokenData = JSON.parse(tokenText); } catch {}

        if (!tokenRes.ok) {
          throw new Error(`Token failed ${tokenRes.status}: ${tokenText}`);
        }

        const userRes = await fetch('https://discord.com/api/users/@me', {
          headers: { authorization: `Bearer ${tokenData.access_token}` }
        });

        const userText = await userRes.text();
        let user = {};
        try { user = JSON.parse(userText); } catch {}

        if (!userRes.ok) {
          throw new Error(`User failed ${userRes.status}: ${userText}`);
        }

        const payload = {
          discordId: String(user.id || ''),
          username: user.discriminator && user.discriminator !== '0'
            ? `${user.username}#${user.discriminator}`
            : String(user.username || 'Discord'),
          avatar: user.avatar || '',
          exp: Math.floor(Date.now() / 1000) + 86400
        };

        const jwt = encodeBase64Url(JSON.stringify(payload));
        const headers = new Headers();
        headers.set(
          'Set-Cookie',
          `discord_auth=${jwt}; Path=/; HttpOnly; Secure; SameSite=Lax; Max-Age=86400`
        );
        headers.set('Location', '/');
        return new Response(null, { status: 302, headers });
      } catch (e) {
        return json({ ok: false, error: e.message || String(e) }, 500, request);
      }
    }

    if (url.pathname === '/api/me') {
      try {
        const jwt = getCookie(request.headers.get('cookie') || '', 'discord_auth');
        if (!jwt) return json({ ok: false }, 401, request);

        const payload = JSON.parse(decodeBase64Url(jwt));
        if (!payload.exp || payload.exp < Math.floor(Date.now() / 1000)) {
          return json({ ok: false, error: 'expired' }, 401, request);
        }

        return json({
          ok: true,
          discordId: payload.discordId || '',
          username: payload.username || 'Discord'
        }, 200, request);
      } catch {
        return json({ ok: false, error: 'invalid' }, 401, request);
      }
    }

    if (
      (url.pathname === '/api/upload' ||
       url.pathname === '/upload' ||
       url.pathname === '/api/contracts/create') &&
      request.method === 'POST'
    ) {
      try {
        const parsed = await parseIncomingRequest(request);
        const apiMode = url.pathname === '/api/contracts/create';

        if (apiMode) {
          requireApiToken(request, env);
        }

        const result = await processContract(env, parsed.payload, parsed.images);

        return json(result, 200, request);
      } catch (e) {
        return json({ ok: false, error: e.message || String(e) }, 400, request);
      }
    }

    return new Response('Not found', {
      status: 404,
      headers: corsHeaders(request)
    });
  }
};

async function parseIncomingRequest(request) {
  let payload = {};
  let images = [];

  const contentType = request.headers.get('content-type') || '';

  if (contentType.includes('multipart/form-data')) {
    const form = await request.formData();

    const payloadRaw = String(form.get('payload') || '{}');
    try {
      payload = JSON.parse(payloadRaw);
    } catch {
      throw new Error('Invalid payload JSON');
    }

    const filesCount = Number(form.get('files_count') || 0);
    for (let i = 0; i < filesCount; i++) {
      const idx = String(i);
      images.push({
        base64: String(form.get('file_' + idx + '_base64') || ''),
        filename: String(form.get('file_' + idx + '_filename') || ('upload_' + (i + 1) + '.jpg')),
        mime: String(form.get('file_' + idx + '_mime') || 'image/jpeg'),
        role: String(form.get('file_' + idx + '_role') || '')
      });
    }
  } else {
    const body = await request.json();
    payload = body && body.payload ? body.payload : {};
    images = Array.isArray(body && body.files) ? body.files : [];
  }

  return { payload, images };
}

function requireApiToken(request, env) {
  const token = request.headers.get('x-api-token') || '';
  if (!env.API_SHARED_TOKEN) {
    throw new Error('API_SHARED_TOKEN is not configured');
  }
  if (!token || token !== env.API_SHARED_TOKEN) {
    throw new Error('Unauthorized API request');
  }
}

async function processContract(env, payload, images) {
  const discordId = String(payload.discordId || '').trim();
  const contractType = String(payload.contractType || '').trim();
  const fields = payload && typeof payload.fields === 'object' && payload.fields ? payload.fields : {};

  if (!discordId) throw new Error('Fill Discord ID');
  if (!/^\d{10,25}$/.test(discordId)) throw new Error('Discord ID looks invalid');
  if (!contractType) throw new Error('Select contract type');

  const channelId = CHANNELS[contractType];
  if (!channelId) throw new Error('Unknown contractType: ' + contractType);

  validateContractSimple(contractType, fields, images);

  const mention = `<@${discordId}>`;
  const built = buildDiscordMessage(mention, contractType, fields, images);

  const parsed = await sendToDiscord(env, String(channelId), built, images);

  const mainMsg = parsed && parsed.mainMessage ? parsed.mainMessage : null;
  const filesMsg = parsed && parsed.filesMessage ? parsed.filesMessage : null;

  const msgId = mainMsg && mainMsg.id ? String(mainMsg.id) : '';
  const attUrls = Array.isArray(filesMsg && filesMsg.attachments)
    ? filesMsg.attachments.map(a => String(a.url || '')).filter(Boolean)
    : [];

  const wnScreenshotsCount = contractType === 'агитации-wn' ? images.length : 0;

  const sheetsResult = await writeToGoogleSheet(env, {
    gmail: '',
    payload,
    files: images.map((x, i) => ({
      filename: makeSafeDiscordFilename(x.filename || ('upload_' + (i + 1) + '.jpg'), i),
      role: x.role || '',
      mime: x.mime || ''
    })),
    channel_id: String(channelId),
    discord_message_id: msgId,
    attachment_urls: attUrls,
    wn_screenshots_count: wnScreenshotsCount,
    status: 'POSTED',
    result: '',
    counted: '0',
    confirm_status: 'PENDING'
  });

  return {
    ok: true,
    rowNumber: sheetsResult && sheetsResult.rowNumber ? sheetsResult.rowNumber : null,
    status: 200,
    discord_message_id: msgId,
    attachment_urls: attUrls,
    wn_screenshots_count: wnScreenshotsCount,
    debug_main_message_id: mainMsg && mainMsg.id ? String(mainMsg.id) : '',
    debug_files_message_id: filesMsg && filesMsg.id ? String(filesMsg.id) : ''
  };
}

async function sendToDiscord(env, channelId, built, images) {
  if (!env.DISCORD_BOT_TOKEN) {
    throw new Error('DISCORD_BOT_TOKEN is not configured');
  }

  const apiUrl = `https://discord.com/api/v10/channels/${channelId}/messages`;
  const botHeaders = {
    authorization: `Bot ${env.DISCORD_BOT_TOKEN}`
  };

  const firstResp = await fetch(apiUrl, {
    method: 'POST',
    headers: {
      ...botHeaders,
      'content-type': 'application/json'
    },
    body: JSON.stringify({
      content: built.content,
      embeds: built.embeds,
      allowed_mentions: { parse: ['users'] }
    })
  });

  const firstText = await firstResp.text();
  let firstParsed = null;
  try { firstParsed = JSON.parse(firstText); } catch {}

  if (!firstResp.ok) {
    throw new Error(`FIRST DISCORD ERROR ${firstResp.status}: ${firstText}`);
  }

  let secondParsed = null;

  if (images.length > 0) {
    const form = new FormData();

    form.append('payload_json', JSON.stringify({
      content: 'Скриншоты по контракту'
    }));

    images.forEach((img, i) => {
      const bytes = base64ToUint8Array(String(img.base64 || ''));
      const safeName = makeSafeDiscordFilename(String(img.filename || `upload_${i + 1}.jpg`), i);
      const blob = new Blob([bytes], { type: String(img.mime || 'image/jpeg') });
      form.append(`files[${i}]`, blob, safeName);
    });

    const secondResp = await fetch(apiUrl, {
      method: 'POST',
      headers: botHeaders,
      body: form
    });

    const secondText = await secondResp.text();
    try { secondParsed = JSON.parse(secondText); } catch {}

    if (!secondResp.ok) {
      throw new Error(`SECOND DISCORD ERROR ${secondResp.status}: ${secondText}`);
    }
  }

  return {
    mainMessage: firstParsed,
    filesMessage: secondParsed
  };
}

async function writeToGoogleSheet(env, data) {
  if (!env.SHEETS_WEBHOOK_URL) {
    throw new Error('SHEETS_WEBHOOK_URL is not configured');
  }
  if (!env.SHEETS_WEBHOOK_TOKEN) {
    throw new Error('SHEETS_WEBHOOK_TOKEN is not configured');
  }

  const res = await fetch(env.SHEETS_WEBHOOK_URL, {
    method: 'POST',
    headers: {
      'content-type': 'application/json'
    },
    body: JSON.stringify({
      token: env.SHEETS_WEBHOOK_TOKEN,
      ...data
    })
  });

  const text = await res.text();
  let parsed = null;
  try { parsed = JSON.parse(text); } catch {}

  if (!res.ok) {
    throw new Error(`Sheets HTTP ${res.status}: ${text}`);
  }
  if (!parsed || !parsed.ok) {
    throw new Error(`Sheets error: ${text}`);
  }

  return parsed;
}

function validateContractSimple(contractType, fields, images) {
  const n = images.length;
  const require = (cond, msg) => { if (!cond) throw new Error(msg); };
  const isInt = (v) => Number.isFinite(v) && Math.floor(v) === v;

  switch (contractType) {
    case 'активация': {
      const price = Number(fields.price);
      require(isInt(price) && price > 0, 'Активация: укажи сумму (целое число > 0).');
      require(price <= 20000, 'Активация: сумма не должна превышать 20000.');
      require(n === 1, 'Активация: можно прикрепить только 1 скриншот.');
      break;
    }
    case 'дары-моря': {
      const fishType = String(fields.fishType || '').trim();
      const fishQty = Number(fields.fishQty);
      require(fishType.length >= 2, 'Дары моря: укажи вид рыбы.');
      require(isInt(fishQty) && fishQty > 0, 'Дары моря: укажи количество (целое число > 0).');
      require(n >= 1 && n <= 10, 'Дары моря: нужно 1..10 скриншотов.');
      break;
    }
    case 'металлургия-сдача': {
      const oreType = String(fields.oreType || '').trim();
      const allowed = ['Железная руда', 'Серебряная руда', 'Медная руда', 'Оловянная руда', 'Золотая руда'];
      require(allowed.includes(oreType), 'Металлургия сдача: выбери руду из списка.');
      require(n === 1, 'Металлургия сдача: можно прикрепить только 1 скриншот.');
      break;
    }
    case 'металлургия-добыча': {
      const keys = ['iron', 'silver', 'copper', 'tin', 'gold'];
      const any = keys.some(k => Number(fields[k]) > 0);
      require(any, 'Металлургия добыча: укажи количество хотя бы для одного ресурса.');
      require(n >= 1 && n <= 10, 'Металлургия добыча: нужно 1..10 скриншотов.');
      break;
    }
    case 'товары': {
      require(n >= 1 && n <= 2, 'Товары: можно 1 или 2 скриншота.');
      break;
    }
    case 'ателье': {
      const total = Number(fields.totalUniforms);
      require(isInt(total) && total > 0, 'Ателье: укажи общее кол-во нашитых форм (целое число > 0).');
      require(n >= 1 && n <= 10, 'Ателье: нужно 1..10 скриншотов.');
      break;
    }
    case 'агитации-маркетплейс': {
      const links = Array.isArray(fields.links) ? fields.links : [];
      require(links.length >= 1, 'Маркетплейс: вставь хотя бы 1 ссылку.');
      require(n === 0, 'Маркетплейс: скриншоты не требуются.');
      break;
    }
    case 'агитации-wn': {
      const category = String(fields.category || '').trim();
      require(['Зеленка(чат)', 'Уведомление'].includes(category), 'WN: выбери категорию.');
      require(n >= 1 && n <= 10, 'WN: нужно 1..10 скриншотов.');
      break;
    }
    case 'тюнинг': {
      require(n === 1, 'Тюнинг: можно прикрепить только 1 скриншот.');
      break;
    }
    default:
      throw new Error('Неизвестный контракт: ' + contractType);
  }
}

function buildDiscordMessage(mention, contractType, fields, images) {
  const contentLines = [mention, `Контракт: ${contractType}`];

  const embed = {
    title: `Отчёт: ${contractType}`,
    color: 0x9b7bff,
    fields: [],
    footer: { text: 'Auto report' },
    timestamp: new Date().toISOString()
  };

  const addField = (name, value, inline = false) => {
    const v = String(value ?? '').trim() || '-';
    embed.fields.push({
      name: String(name).slice(0, 256),
      value: v.slice(0, 1024),
      inline: !!inline
    });
  };

  if (contractType === 'активация') {
    addField('Сумма', String(Number(fields.price || 0)), true);
  } else if (contractType === 'дары-моря') {
    addField('Рыба', String(fields.fishType || '').trim() || '-', true);
    addField('Количество', String(Number(fields.fishQty || 0)), true);
  } else if (contractType === 'металлургия-сдача') {
    addField('Руда', String(fields.oreType || '').trim() || '-', true);
  } else if (contractType === 'металлургия-добыча') {
    const map = [
      ['iron', 'Железо'],
      ['silver', 'Серебро'],
      ['copper', 'Медь'],
      ['tin', 'Олово'],
      ['gold', 'Золото']
    ];
    map.forEach(([k, label]) => {
      const v = Number(fields[k] || 0);
      if (v > 0) addField(label, String(v), true);
    });
  } else if (contractType === 'товары') {
    const hasDelivery = images.some(x => String(x.role || '') === 'delivery');
    const hasLoading = images.some(x => String(x.role || '') === 'loading');
    addField('Сдача', hasDelivery ? 'YES' : '', true);
    addField('Погрузка', hasLoading ? 'YES' : '', true);
  } else if (contractType === 'ателье') {
    addField('Нашито форм (всего)', String(Number(fields.totalUniforms || 0)), true);
  } else if (contractType === 'агитации-маркетплейс') {
    const links = Array.isArray(fields.links) ? fields.links : [];
    addField('Ссылок', String(links.length), true);
    addField('Ссылки', links.slice(0, 10).join('\n') || '-', false);
  } else if (contractType === 'агитации-wn') {
    addField('Категория', String(fields.category || '').trim() || '-', true);
    addField('Скриншотов', String(images.length), true);
  } else if (contractType === 'тюнинг') {
    addField('Скриншот', images.length ? 'YES' : '', true);
  }

  if (images.length) {
    const list = images.map((img, i) => {
      const role = String(img.role || '').trim();
      const name = makeSafeDiscordFilename(String(img.filename || `upload_${i + 1}.jpg`), i);
      return role ? `${i + 1}. ${role} — ${name}` : `${i + 1}. ${name}`;
    }).join('\n');

    addField('Скриншоты', list, false);
  }

  return {
    content: contentLines.join('\n'),
    embeds: [embed]
  };
}

function json(data, status = 200, request = null) {
  return new Response(JSON.stringify(data), {
    status,
    headers: {
      'content-type': 'application/json; charset=UTF-8',
      ...(request ? corsHeaders(request) : {})
    }
  });
}

function corsHeaders(request) {
  const origin = request.headers.get('origin') || '*';
  return {
    'Access-Control-Allow-Origin': origin,
    'Access-Control-Allow-Methods': 'GET,POST,OPTIONS',
    'Access-Control-Allow-Headers': 'Content-Type, x-api-token',
    'Access-Control-Allow-Credentials': 'true'
  };
}

function getCookie(cookieHeader, name) {
  const parts = cookieHeader.split(';').map(x => x.trim());
  for (const p of parts) {
    const i = p.indexOf('=');
    if (i === -1) continue;
    const k = p.slice(0, i);
    const v = p.slice(i + 1);
    if (k === name) return v;
  }
  return '';
}

function encodeBase64Url(str) {
  return btoa(unescape(encodeURIComponent(str)))
    .replace(/\+/g, '-')
    .replace(/\//g, '_')
    .replace(/=+$/g, '');
}

function decodeBase64Url(str) {
  const pad = str.length % 4 === 0 ? '' : '='.repeat(4 - (str.length % 4));
  const base64 = str.replace(/-/g, '+').replace(/_/g, '/') + pad;
  return decodeURIComponent(escape(atob(base64)));
}

function base64ToUint8Array(base64) {
  const bin = atob(base64);
  const arr = new Uint8Array(bin.length);
  for (let i = 0; i < bin.length; i++) arr[i] = bin.charCodeAt(i);
  return arr;
}

function makeSafeDiscordFilename(name, index = 0) {
  const extMatch = String(name).match(/\.([a-zA-Z0-9]+)$/);
  const ext = extMatch ? `.${extMatch[1].toLowerCase()}` : '.jpg';

  const base = String(name)
    .replace(/\.[^/.]+$/, '')
    .normalize('NFKD')
    .replace(/[^\w.-]+/g, '_')
    .replace(/_+/g, '_')
    .replace(/^_+|_+$/g, '')
    .slice(0, 30) || 'upload';

  return `${base}_${index + 1}${ext}`;
}

    const buf = await file.arrayBuffer();
    const hashBuf = await crypto.subtle.digest('SHA-256', buf);

      const dtClip = e.clipboardData;
      if (!dtClip) return;

      const images = [];
      const items = dtClip.items ? [...dtClip.items] : [];
      for (const it of items) {
        if (it.kind === 'file' && it.type && it.type.startsWith('image/')) {
          const f = it.getAsFile();
          if (f) images.push(f);
        }
      }
      if (!images.length && dtClip.files && dtClip.files.length) {
        const f = dtClip.files[0];
        if (f && f.type && f.type.startsWith('image/')) images.push(f);
      }
      if (!images.length) return;

      e.preventDefault();
      const ct = ctSel.value;

      function getInput(id){
        const el = document.getElementById(id);
        return (el && el.type === 'file') ? el : null;
      }

      function pickTargetIdForGoodsByRule(){
        if (hoveredFileTarget && (hoveredFileTarget === 'fileDelivery' || hoveredFileTarget === 'fileLoading')) {
          return hoveredFileTarget;
        }
        const d = getInput('fileDelivery');
        const l = getInput('fileLoading');
        if (d && (!d.files || d.files.length === 0)) return 'fileDelivery';
        if (l && (!l.files || l.files.length === 0)) return 'fileLoading';
        return lastFileTarget || 'fileLoading';
      }

      function pickTargetIdDefault(){
        if (ct === 'товары') return pickTargetIdForGoodsByRule();
        return hoveredFileTarget || lastFileTarget || 'filesMain';
      }

      for (const imgFile of images) {
        const targetId = pickTargetIdDefault();
        const inp = getInput(targetId);
        if (!inp) continue;

        const hset = getHashSet(inp.id);
        const hex = await sha256Hex(imgFile);
        if (hset.has(hex)) continue;
        hset.add(hex);

        const dt = new DataTransfer();
        const existing = inp.files ? [...inp.files] : [];
        if (inp.multiple) {
          existing.forEach(f => dt.items.add(f));
          dt.items.add(imgFile);
        } else {
          dt.items.add(imgFile);
        }

        inp.files = dt.files;
        inp.dispatchEvent(new Event('change', { bubbles: true }));

        if (ct === 'товары') {
          if (targetId === 'fileDelivery') rememberOrder('delivery');
          if (targetId === 'fileLoading') rememberOrder('loading');
          lastFileTarget = targetId;
        }
      }
    } catch (err) {
      console.error('ERROR:', err && err.message ? err.message : String(err));
    }
  });

  function asciiFilename(originalName){
    const base = (originalName || 'upload').replace(/\.[^/.]+$/, '');
    const ascii = base.normalize('NFKD')
      .replace(/[^\w.-]+/g, '_')
      .replace(/_+/g, '_')
      .replace(/^_+|_+$/g, '')
      .slice(0, 40) || 'upload';
    return ascii + '.jpg';
  }

  function blobToDataURL(blob){
    return new Promise((resolve, reject) => {
      const r = new FileReader();
      r.onload = () => resolve(r.result);
      r.onerror = reject;
      r.readAsDataURL(blob);
    });
  }

  function resizeToJpegBlob(file, maxSide, quality){
    return new Promise((resolve, reject) => {
      const img = new Image();
      const url = URL.createObjectURL(file);

      img.onload = () => {
        URL.revokeObjectURL(url);
        let w = img.naturalWidth || img.width;
        let h = img.naturalHeight || img.height;
        const scale = Math.min(1, maxSide / Math.max(w, h));
        const nw = Math.max(1, Math.round(w * scale));
        const nh = Math.max(1, Math.round(h * scale));
        const canvas = document.createElement('canvas');
        canvas.width = nw;
        canvas.height = nh;
        const ctx = canvas.getContext('2d', { alpha:false });
        ctx.drawImage(img, 0, 0, nw, nh);
        canvas.toBlob(
          (blob) => blob ? resolve(blob) : reject(new Error('toBlob failed')),
          'image/jpeg',
          quality
        );
      };

      img.onerror = () => {
        URL.revokeObjectURL(url);
        reject(new Error('Image decode failed'));
      };

      img.src = url;
    });
  }

  async function fileToJpegBase64Obj(file){
    const resizedBlob = await resizeToJpegBlob(file, MAX_SIDE, JPEG_QUALITY);
    const dataUrl = await blobToDataURL(resizedBlob);
    const base64 = String(dataUrl).split(',')[1];
    return { base64, filename: asciiFilename(file.name), mime: 'image/jpeg' };
  }

  async function send(){
    const btn = document.getElementById('sendBtn');
    if (btn.disabled) return;
  
    const COOLDOWN_MS = 10000;
    const originalText = btn.dataset.originalText || btn.textContent;
    btn.dataset.originalText = originalText;
  
    btn.disabled = true;
    setSending(true);
  
    function unlockNow(){
      btn.disabled = false;
      btn.textContent = originalText;
    }
  
    function startCooldown(){
      let left = Math.ceil(COOLDOWN_MS / 1000);
      btn.disabled = true;
      btn.textContent = originalText + ' (' + left + ')';
  
      const timer = setInterval(() => {
        left -= 1;
        if (left <= 0) {
          clearInterval(timer);
          btn.disabled = false;
          btn.textContent = originalText;
        } else {
          btn.textContent = originalText + ' (' + left + ')';
        }
      }, 1000);
    }
  
    try {
      const collected = await collectPayloadAndFiles();
  
      const form = new FormData();
      form.append('payload', JSON.stringify(collected.payload));
      form.append('files_count', String(collected.files.length));
  
      collected.files.forEach((f, i) => {
        const idx = String(i);
        form.append('file_' + idx + '_base64', String(f.base64 || ''));
        form.append('file_' + idx + '_filename', String(f.filename || ('upload_' + (i + 1) + '.jpg')));
        form.append('file_' + idx + '_mime', String(f.mime || 'image/jpeg'));
        form.append('file_' + idx + '_role', String(f.role || ''));
      });
  
      const res = await fetch('/api/upload', {
        method: 'POST',
        body: form
      });
  
      const raw = await res.text();
  
      let data = null;
      try {
        data = raw ? JSON.parse(raw) : null;
      } catch (e) {
        throw new Error('Сервер вернул не JSON: ' + raw.slice(0, 300));
      }
  
      if (res.ok && data && data.ok) {
        setOut('', true);
        startCooldown();
      } else {
        setOut((data && (data.error || data.body)) || ('HTTP ' + res.status), false);
        unlockNow();
      }
    } catch(e){
      console.error('SEND ERROR:', e);
      setOut(e && e.message ? e.message : String(e), false);
      unlockNow();
    }
  }

  document.getElementById('sendBtn').addEventListener('click', send);
  setOut('', null);
